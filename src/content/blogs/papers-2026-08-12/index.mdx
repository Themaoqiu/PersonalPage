  title={"Macaron-V1: Towards Open Continual Learning with Self-Improvement and Mixture-of-LoRA"}
  arxivUrl={"https://arxiv.org/abs/2608.09819"}

<PaperCard
  title={"SWE-Bench ProMax: Benchmarking Agents on Large-Scale Multilingual Code Refactoring"}
  arxivUrl={"https://arxiv.org/abs/2608.09802"}
  hjfyUrl={"https://hjfy.top/arxiv/2608.09802"}
  summary={"这篇论文提出 SWE-Bench ProMax，一个面向大规模、多语言代码重构的基准，用来评测软件工程智能体在长程、跨文件修改任务上的能力。作者从真实提交中精选了 170 个实例，覆盖 Python、Java、TypeScript、Go、C、C++ 和 Rust，并通过多阶段人工整理重写题面、审查测试，尽量去除过窄或过宽的评测缺陷。基准任务平均涉及 11.4 个被修改文件和 261.6 行代码，明显高于现有基准的规模。实验显示，在两种 agent scaffold 下最强模型的解决率也只有 41.2%，说明该基准仍然具有较强挑战性。"}
/>

<PaperCard
  title={"Ouroboros: A Self-Developing Frontier Coding Agent with Reviewed Core Evolution"}
  arxivUrl={"https://arxiv.org/abs/2608.08311"}

<PaperCard
  title={"Agent Memory Distillation: Empowering Small LLM Agents with Hierarchical Teacher Memory"}
  arxivUrl={"https://arxiv.org/abs/2608.07169"}
  hjfyUrl={"https://hjfy.top/arxiv/2608.07169"}
  summary={"本文提出 Agent Memory Distillation（AMD），用层次化记忆把大模型教师代理的成功轨迹知识迁移给小模型代理，而且不需要额外训练。它构造了三类记忆：Workflow 记忆描述任务级策略，Subtask 记忆提供中粒度行为示例，Function 记忆总结函数调用约定和常见错误。前两类记忆在任务开始时主动注入，Function 记忆则在工具调用出错时按需检索，以提升工具使用稳定性。作者在 AppWorld、BFCL V3 和 ToolSandbox 上用 4B 到 8B 的学生模型验证，平均分别提升 27.2、11.2 和 3.4 个百分点，并且整体优于已有记忆式基线。"}
